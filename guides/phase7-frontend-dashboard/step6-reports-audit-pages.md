# Phase 7 - Step 6: Reports Page + Audit Logs Page

## Why Are We Doing This?

Two remaining pages complete the frontend:

1. **Reports page** — lists all generated reports with status, risk score, and "View" / "Export PDF" actions. The report detail page shows the full executive summary, action plans, and risk breakdown.

2. **Audit Logs page** — displays the compliance trail. Only ADMIN and HR_MANAGER can access it. Shows who did what, when, with filtering by action type, entity type, user, and date range.

---

## What We're Building

```
frontend/src/
  app/(dashboard)/
    reports/
      page.tsx               ← Reports list with status tags
      [id]/page.tsx          ← Report detail: summary + action plans + risk
    audit-logs/
      page.tsx               ← Audit log table with filters + pagination
```

## The Steps

### Step A: Create the Reports list page

Create `frontend/src/app/(dashboard)/reports/page.tsx`:

```typescript
'use client';

import { useState, useEffect } from 'react';
import { Typography, Table, Tag, Button, Space } from 'antd';
import { EyeOutlined, FilePdfOutlined } from '@ant-design/icons';
import { useRouter } from 'next/navigation';
import api from '@/lib/api';
import type { Report } from '@/types';
import type { ColumnsType } from 'antd/es/table';
import { getRiskLevel, RISK_COLORS } from '@/lib/constants';

const { Title } = Typography;

const STATUS_COLORS: Record<string, string> = {
  COMPLETED: 'green',
  GENERATING: 'blue',
  FAILED: 'red',
};

export default function ReportsPage() {
  const router = useRouter();
  const [reports, setReports] = useState<Report[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    api.get<Report[]>('/reports')
      .then(({ data }) => setReports(data))
      .catch(() => setReports([]))
      .finally(() => setLoading(false));
  }, []);

  const handleExportPdf = async (reportId: string) => {
    try {
      const response = await api.get(`/reports/${reportId}/pdf`, {
        responseType: 'blob',
      });
      const url = window.URL.createObjectURL(new Blob([response.data]));
      const link = document.createElement('a');
      link.href = url;
      link.download = `report-${reportId}.pdf`;
      link.click();
      window.URL.revokeObjectURL(url);
    } catch {
      // PDF export may not be available until Phase 6
    }
  };

  const columns: ColumnsType<Report> = [
    {
      title: 'Team',
      dataIndex: ['team', 'name'],
      sorter: (a, b) => (a.team?.name ?? '').localeCompare(b.team?.name ?? ''),
    },
    {
      title: 'Generated',
      dataIndex: 'createdAt',
      render: (d: string) => new Date(d).toLocaleString(),
      sorter: (a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime(),
      defaultSortOrder: 'descend',
    },
    {
      title: 'Status',
      dataIndex: 'status',
      render: (s: string) => <Tag color={STATUS_COLORS[s]}>{s}</Tag>,
      filters: [
        { text: 'Completed', value: 'COMPLETED' },
        { text: 'Generating', value: 'GENERATING' },
        { text: 'Failed', value: 'FAILED' },
      ],
      onFilter: (value, record) => record.status === value,
    },
    {
      title: 'Risk Score',
      dataIndex: 'riskScore',
      render: (score: number | null) => {
        if (score === null) return '—';
        const level = getRiskLevel(score);
        return <Tag color={RISK_COLORS[level]}>{Math.round(score * 100)}%</Tag>;
      },
      sorter: (a, b) => (a.riskScore ?? 0) - (b.riskScore ?? 0),
    },
    {
      title: 'Generated By',
      dataIndex: ['generatedByUser', 'email'],
    },
    {
      title: 'Actions',
      key: 'actions',
      render: (_, record) => (
        <Space>
          <Button size="small" icon={<EyeOutlined />} onClick={() => router.push(`/reports/${record.id}`)}>
            View
          </Button>
          {record.status === 'COMPLETED' && (
            <Button size="small" icon={<FilePdfOutlined />} onClick={() => handleExportPdf(record.id)}>
              PDF
            </Button>
          )}
        </Space>
      ),
    },
  ];

  return (
    <div>
      <Title level={3}>Reports</Title>
      <Table columns={columns} dataSource={reports} rowKey="id" loading={loading} pagination={{ pageSize: 10 }} />
    </div>
  );
}
```

### Step B: Create the Report Detail page

Create `frontend/src/app/(dashboard)/reports/[id]/page.tsx`:

```typescript
'use client';

import { useState, useEffect } from 'react';
import { useParams, useRouter } from 'next/navigation';
import { Typography, Card, Tag, Breadcrumb, Spin, Row, Col, Descriptions, Button } from 'antd';
import { FilePdfOutlined } from '@ant-design/icons';
import api from '@/lib/api';
import type { Report } from '@/types';
import { getRiskLevel, RISK_COLORS, RISK_LABELS } from '@/lib/constants';
import RiskScoreCard from '@/components/dashboard/risk-score-card';
import ExecutiveSummary from '@/components/dashboard/executive-summary';
import ActionPlanCards from '@/components/dashboard/action-plan-cards';

const { Title } = Typography;

export default function ReportDetailPage() {
  const { id } = useParams<{ id: string }>();
  const router = useRouter();
  const [report, setReport] = useState<Report | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    api.get<Report>(`/reports/${id}`)
      .then(({ data }) => setReport(data))
      .finally(() => setLoading(false));
  }, [id]);

  if (loading) return <Spin size="large" style={{ display: 'block', margin: '48px auto' }} />;
  if (!report) return <Title level={4}>Report not found</Title>;

  return (
    <div>
      <Breadcrumb items={[
        { title: <a onClick={() => router.push('/reports')}>Reports</a> },
        { title: report.team?.name ?? 'Report' },
      ]} />

      <div style={{ display: 'flex', justifyContent: 'space-between', marginTop: 16, marginBottom: 16 }}>
        <Title level={3} style={{ margin: 0 }}>
          Report: {report.team?.name}
        </Title>
        {report.status === 'COMPLETED' && (
          <Button type="primary" icon={<FilePdfOutlined />}>Export PDF</Button>
        )}
      </div>

      <Descriptions bordered size="small" column={3} style={{ marginBottom: 16 }}>
        <Descriptions.Item label="Status">
          <Tag color={report.status === 'COMPLETED' ? 'green' : report.status === 'FAILED' ? 'red' : 'blue'}>
            {report.status}
          </Tag>
        </Descriptions.Item>
        <Descriptions.Item label="Generated">
          {new Date(report.createdAt).toLocaleString()}
        </Descriptions.Item>
        <Descriptions.Item label="Date Range">
          {new Date(report.dateRangeStart).toLocaleDateString()} — {new Date(report.dateRangeEnd).toLocaleDateString()}
        </Descriptions.Item>
        <Descriptions.Item label="Model Version">{report.modelVersion ?? '—'}</Descriptions.Item>
      </Descriptions>

      <Row gutter={16}>
        <Col xs={24} sm={8}>
          <RiskScoreCard score={report.riskScore} />
        </Col>
        <Col xs={24} sm={16}>
          <ExecutiveSummary summary={report.summaryText} />
        </Col>
      </Row>

      {report.actionPlans && report.actionPlans.length > 0 && (
        <div style={{ marginTop: 16 }}>
          <ActionPlanCards actionPlans={report.actionPlans} />
        </div>
      )}
    </div>
  );
}
```

### Step C: Create the Audit Logs page

Create `frontend/src/app/(dashboard)/audit-logs/page.tsx`:

```typescript
'use client';

import { useState, useEffect, useCallback } from 'react';
import { Typography, Table, Tag, Select, DatePicker, Space, Card } from 'antd';
import dayjs from 'dayjs';
import api from '@/lib/api';
import type { AuditLog, AuditAction, PaginatedResponse } from '@/types';
import type { ColumnsType } from 'antd/es/table';

const { Title } = Typography;
const { RangePicker } = DatePicker;

const ACTION_COLORS: Record<AuditAction, string> = {
  CREATE: 'green',
  UPDATE: 'blue',
  DELETE: 'red',
  GENERATE_REPORT: 'purple',
  EXPORT_PDF: 'cyan',
  LOGIN: 'default',
};

export default function AuditLogsPage() {
  const [logs, setLogs] = useState<AuditLog[]>([]);
  const [loading, setLoading] = useState(true);
  const [total, setTotal] = useState(0);
  const [page, setPage] = useState(1);
  const [limit] = useState(20);
  const [actionFilter, setActionFilter] = useState<string | undefined>();
  const [entityFilter, setEntityFilter] = useState<string | undefined>();
  const [dateRange, setDateRange] = useState<[string, string] | null>(null);

  const fetchLogs = useCallback(async () => {
    setLoading(true);
    const params: Record<string, string | number> = { page, limit };
    if (actionFilter) params.action = actionFilter;
    if (entityFilter) params.entityType = entityFilter;
    if (dateRange) {
      params.from = dateRange[0];
      params.to = dateRange[1];
    }

    try {
      const { data } = await api.get<PaginatedResponse<AuditLog>>('/audit-logs', { params });
      setLogs(data.data);
      setTotal(data.meta.total);
    } finally {
      setLoading(false);
    }
  }, [page, limit, actionFilter, entityFilter, dateRange]);

  useEffect(() => { fetchLogs(); }, [fetchLogs]);

  const columns: ColumnsType<AuditLog> = [
    {
      title: 'Time',
      dataIndex: 'createdAt',
      render: (d: string) => new Date(d).toLocaleString(),
      width: 180,
    },
    {
      title: 'User',
      dataIndex: ['user', 'email'],
      width: 200,
    },
    {
      title: 'Action',
      dataIndex: 'action',
      render: (action: AuditAction) => (
        <Tag color={ACTION_COLORS[action]}>{action}</Tag>
      ),
      width: 150,
    },
    {
      title: 'Entity',
      dataIndex: 'entityType',
      render: (type: string) => <Tag>{type}</Tag>,
      width: 120,
    },
    {
      title: 'Entity ID',
      dataIndex: 'entityId',
      render: (id: string | null) => id ? `...${id.slice(-8)}` : '—',
      width: 120,
    },
    {
      title: 'IP Address',
      dataIndex: 'ipAddress',
      width: 130,
    },
  ];

  return (
    <div>
      <Title level={3}>Audit Logs</Title>

      <Card size="small" style={{ marginBottom: 16 }}>
        <Space wrap>
          <Select
            placeholder="Filter by action"
            allowClear
            style={{ width: 180 }}
            onChange={(v) => { setActionFilter(v); setPage(1); }}
            options={[
              { label: 'CREATE', value: 'CREATE' },
              { label: 'UPDATE', value: 'UPDATE' },
              { label: 'DELETE', value: 'DELETE' },
              { label: 'GENERATE_REPORT', value: 'GENERATE_REPORT' },
              { label: 'EXPORT_PDF', value: 'EXPORT_PDF' },
              { label: 'LOGIN', value: 'LOGIN' },
            ]}
          />
          <Select
            placeholder="Filter by entity"
            allowClear
            style={{ width: 160 }}
            onChange={(v) => { setEntityFilter(v); setPage(1); }}
            options={[
              { label: 'USER', value: 'USER' },
              { label: 'TEAM', value: 'TEAM' },
              { label: 'EMPLOYEE', value: 'EMPLOYEE' },
              { label: 'REPORT', value: 'REPORT' },
            ]}
          />
          <RangePicker
            onChange={(dates) => {
              if (dates && dates[0] && dates[1]) {
                setDateRange([dates[0].toISOString(), dates[1].toISOString()]);
              } else {
                setDateRange(null);
              }
              setPage(1);
            }}
          />
        </Space>
      </Card>

      <Table
        columns={columns}
        dataSource={logs}
        rowKey="id"
        loading={loading}
        pagination={{
          current: page,
          pageSize: limit,
          total,
          onChange: setPage,
          showTotal: (total) => `Total ${total} entries`,
        }}
      />
    </div>
  );
}
```

## How to Verify It Worked

1. Navigate to Reports page → empty or shows existing reports (depending on Phase 6 status)
2. If reports exist, click "View" → see detail page with summary + action plans
3. Navigate to Audit Logs → see entries from previous mutations
4. Filter by action "CREATE" → only CREATE entries shown
5. Filter by entity "TEAM" → only team-related entries
6. Pagination works — change page, total updates
7. Login as TEAM_MANAGER → "Audit Logs" not in sidebar menu
8. Login as VIEWER → "Audit Logs" not in sidebar menu
